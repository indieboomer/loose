cmake_minimum_required(VERSION 3.25)
project(Loose VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(FETCHCONTENT_QUIET OFF)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLM_ENABLE_CXX_20 ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GENERATE_DEBUG_SYMBOLS ON CACHE BOOL "" FORCE)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
set(OVERRIDE_CXX_FLAGS OFF CACHE BOOL "" FORCE)
set(CROSS_PLATFORM_DETERMINISTIC ON CACHE BOOL "" FORCE)
set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_HELLO_WORLD OFF CACHE BOOL "" FORCE)
set(BUILD_PERFORMANCE_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_VIEWER OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES ON CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(VULKAN_HEADERS_ENABLE_MODULE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
FetchContent_Declare(jolt GIT_REPOSITORY https://github.com/jrouwe/JoltPhysics.git GIT_TAG v5.3.0 GIT_SHALLOW TRUE SOURCE_SUBDIR Build)
FetchContent_Declare(vkheaders GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git GIT_TAG v1.3.296 GIT_SHALLOW TRUE)
FetchContent_Declare(glslang GIT_REPOSITORY https://github.com/KhronosGroup/glslang.git GIT_TAG 15.0.0 GIT_SHALLOW TRUE)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.5 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(glfw glm jolt vkheaders glslang imgui)
add_library(loose_ui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp)
target_include_directories(loose_ui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_compile_definitions(loose_ui PUBLIC VK_NO_PROTOTYPES IMGUI_IMPL_VULKAN_NO_PROTOTYPES GLFW_INCLUDE_NONE)
target_link_libraries(loose_ui PUBLIC glfw Vulkan::Headers)
add_library(loose_core STATIC src/level.cpp src/orientation.cpp src/physics.cpp src/game.cpp)
target_include_directories(loose_core PUBLIC src)
target_compile_definitions(loose_core PUBLIC GLM_FORCE_DEPTH_ZERO_TO_ONE GLM_FORCE_RADIANS GLM_ENABLE_EXPERIMENTAL)
target_link_libraries(loose_core PUBLIC glm::glm Jolt)
add_executable(loose src/main.cpp src/renderer.cpp src/ui.cpp src/verification.cpp)
target_link_libraries(loose PRIVATE loose_core loose_ui)
target_compile_definitions(loose PRIVATE LOOSE_SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}/shaders")
if(MSVC)
  target_compile_options(loose PRIVATE /W4 /EHsc)
  target_compile_options(loose_core PRIVATE /W4 /EHsc)
endif()
set(SHADERS geometry.vert geometry.frag shadow.vert fullscreen.vert lighting.frag post.frag)
foreach(SHADER IN LISTS SHADERS)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders
    COMMAND $<TARGET_FILE:glslang-standalone> -V --target-env vulkan1.1 -o ${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER} glslang-standalone VERBATIM)
  list(APPEND SHADER_OUTPUTS ${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv)
endforeach()
add_custom_target(loose_shaders DEPENDS ${SHADER_OUTPUTS})
add_dependencies(loose loose_shaders)
add_custom_command(TARGET loose POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders $<TARGET_FILE_DIR:loose>/shaders)
enable_testing()
add_executable(loose_tests tests/core_tests.cpp)
target_link_libraries(loose_tests PRIVATE loose_core)
add_test(NAME mechanics COMMAND loose_tests)
